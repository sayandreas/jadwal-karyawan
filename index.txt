    function normalizeState(st) {
      const def = createInitialState();
      st.settings = Object.assign({}, def.settings, st.settings || {});
      if (!Array.isArray(st.settings.lateTiers) || st.settings.lateTiers.length === 0) {
        st.settings.lateTiers = def.settings.lateTiers;
      }
      st.settings.shiftHours = Object.assign({}, def.settings.shiftHours, st.settings.shiftHours || {});
      st.settings.forbiddenOffDays = Object.assign({}, def.settings.forbiddenOffDays, st.settings.forbiddenOffDays || {});
      // Pastikan hari Sabtu (6) selalu dilarang Off di semua section
      ['BAR', 'Server', 'Kitchen'].forEach(sec => {
        if (!Array.isArray(st.settings.forbiddenOffDays[sec])) st.settings.forbiddenOffDays[sec] = [6];
        if (!st.settings.forbiddenOffDays[sec].includes(6)) st.settings.forbiddenOffDays[sec].push(6);
      });
      st.periods.forEach(p => {
        if (!Array.isArray(p.requestOffs)) p.requestOffs = [];
        p.days.forEach(d => {
          if (d.dow === undefined && d.iso) {
            d.dow = new Date(d.iso + 'T00:00:00').getDay();
          }
        });
      });
      return st;
    }

    function loadState() {
      try {
        const raw = localStorage.getItem(STORAGE_KEY);

          if (parsed && parsed.periods && parsed.employees) return normalizeState(parsed);
        }
      } catch (e) {
        console.warn('Gagal membaca localStorage:', e);
      }
      return normalizeState(createInitialState());
    }

    function ensureEmployeeRecord(period, empId) {
      if (!period.records[empId]) {
        const len = period.days.length;
        period.records[empId] = {

    function formatRupiah(num) {
      const val = Math.round(num || 0);
      const prefix = val < 0 ? '-Rp ' : 'Rp ';
      return prefix + Math.abs(val).toLocaleString('id-ID');
    }

    function calculateLatenessFine(totalMinutes) {
      const m = Number(totalMinutes) || 0;
      if (m <= 0) return 0;
      const tiers = [...(appState.settings.lateTiers || [])].sort((a, b) => Number(a.maxMinutes) - Number(b.maxMinutes));
      if (tiers.length === 0) return 0;
      for (const tier of tiers) {
        if (m <= Number(tier.maxMinutes)) {
          return Number(tier.fine) || 0;
        }
      }
      return Number(tiers[tiers.length - 1].fine) || 0;
    }

    function getLateTierSummaryText() {
      const tiers = [...(appState.settings.lateTiers || [])].sort((a, b) => Number(a.maxMinutes) - Number(b.maxMinutes));
      let prev = 1;
      const parts = ['0m = Rp 0'];
      tiers.forEach((t, idx) => {
        const maxM = Number(t.maxMinutes);
        const fineK = Math.round((Number(t.fine) || 0) / 1000) + 'rb';
        if (idx === tiers.length - 1 && maxM >= 999) {
          parts.push(`&gt;${prev - 1}m = Rp ${fineK}`);
        } else {
          parts.push(`${prev}–${maxM}m = Rp ${fineK}`);
          prev = maxM + 1;
        }
      });
      return parts.join(' • ');
    }

    function computeEmployeeMetrics(emp, period) {
      const rec = ensureEmployeeRecord(period, emp.id);
      let countP = 0, countMD = 0, countM = 0, countO = 0, countS = 0, countI = 0, countA = 0;

      rec.attendance.forEach(code => {
        if (code === 'P') countP++;
        else if (code === 'MD') countMD++;
        else if (code === 'M') countM++;
        else if (code === 'O') countO++;
        else if (code === 'S') countS++;
        else if (code === 'I') countI++;
        else if (code === 'A') countA++;
      });

      const totalMasuk = countP + countMD + countM;
      const totalOtHours = rec.overtime.reduce((sum, v) => sum + (Number(v) || 0), 0);
      const totalLateMinutes = rec.lateness.reduce((sum, v) => sum + (Number(v) || 0), 0);

      const divisor = Number(period.workdayDivisor) || 26;
      const otRate = Number(appState.settings.overtimeRate) || 15000;
      const alpaRate = Number(appState.settings.alpaFine) || 100000;
      const deductSI = appState.settings.deductSickLeave !== false;

      const dailyRate = emp.baseSalary / divisor;
      const sickLeaveDays = countS + countI;
      const sickLeaveDeduction = deductSI ? Math.round(dailyRate * sickLeaveDays) : 0;
      const alpaDeduction = countA * alpaRate;
      const latenessDeduction = calculateLatenessFine(totalLateMinutes);

      const overtimePay = Math.round(totalOtHours * otRate);
      const incentive = Number(emp.incentive) || 0;

      const totalDeductions = sickLeaveDeduction + alpaDeduction + latenessDeduction;
      const netSalary = emp.baseSalary + overtimePay + incentive - totalDeductions;

      return {
        countP, countMD, countM, countO, countS, countI, countA,
        totalMasuk,
        totalOtHours,
        totalLateMinutes,
        dailyRate: Math.round(dailyRate),
        sickLeaveDays,
        sickLeaveDeduction,
        alpaDeduction,
        latenessDeduction,
        overtimePay,
        incentive,
        totalDeductions,
        netSalary
      };
    }

    function detectScheduleViolations(period) {
      const emps = getPeriodEmployees(period);
      const numDays = period.days.length;
      const maxOff = Number(appState.settings.maxDailyOffPerSection) || 1;
      const reqOffEmpIds = new Set((period.requestOffs || []).map(r => r.empId));

      const mpJumps = [];
      const multiOffs = [];
      const saturdayOffs = [];

      // 1. Cek jumping Malam -> Pagi (M -> P) per karyawan (dikecualikan jika karyawan punya Request Off)
      emps.forEach(emp => {
        const rec = ensureEmployeeRecord(period, emp.id);
        for (let d = 0; d < numDays; d++) {
          if (period.days[d].dow === 6 && rec.attendance[d] === 'O') {
            saturdayOffs.push({
              empId: emp.id,
              empName: emp.name,
              dayIdx: d,
              dayLabel: period.days[d].fullLabel
            });
          }
          if (d > 0 && rec.attendance[d - 1] === 'M' && rec.attendance[d] === 'P') {
            if (!reqOffEmpIds.has(emp.id)) {
              mpJumps.push({
                empId: emp.id,
                empName: emp.name,
                dayIdx: d,
                dayLabel: period.days[d].fullLabel
              });
            }
          }
        }
      });

      // 2. Cek > maxOff orang Off dalam 1 section pada hari yang sama
      const sections = ['BAR', 'Server', 'Kitchen'];
      sections.forEach(sec => {
        const secEmps = emps.filter(e => e.section === sec);
        for (let d = 0; d < numDays; d++) {
          const offEmps = secEmps.filter(e => ensureEmployeeRecord(period, e.id).attendance[d] === 'O');
          if (offEmps.length > maxOff) {
            multiOffs.push({
              section: sec,
              dayIdx: d,
              dayLabel: period.days[d].fullLabel,
              names: offEmps.map(e => e.name),
              empIds: offEmps.map(e => e.id)
            });
          }
        }
      });

      return { mpJumps, multiOffs, saturdayOffs };
    }

    function autoFixScheduleViolations() {
      const period = getActivePeriod();
      const emps = getPeriodEmployees(period);
      const numDays = period.days.length;
      const maxOff = Number(appState.settings.maxDailyOffPerSection) || 1;
      const reqOffEmpIds = new Set((period.requestOffs || []).map(r => r.empId));
      let fixedCount = 0;

      // 1. Hapus semua Off di hari Sabtu (tidak boleh ada yang Off hari Sabtu)
      emps.forEach(emp => {
        const rec = ensureEmployeeRecord(period, emp.id);
        const secEmps = emps.filter(e => e.section === emp.section);
        for (let d = 0; d < numDays; d++) {
          if (period.days[d].dow === 6 && rec.attendance[d] === 'O') {
            const prevCode = d > 0 ? rec.attendance[d - 1] : '';
            const pCount = secEmps.filter(e => ensureEmployeeRecord(period, e.id).attendance[d] === 'P').length;
            rec.attendance[d] = (prevCode !== 'M' && pCount === 0) ? 'P' : 'M';
            fixedCount++;
          }
        }
      });

      // 2. Perbaiki M -> P (kecuali karyawan Request Off)
      emps.forEach(emp => {
        if (reqOffEmpIds.has(emp.id)) return;
        const rec = ensureEmployeeRecord(period, emp.id);
        const secEmps = emps.filter(e => e.section === emp.section);
        const forbiddenDays = appState.settings.forbiddenOffDays[emp.section] || [6];
        for (let d = 1; d < numDays; d++) {
          if (rec.attendance[d - 1] === 'M' && rec.attendance[d] === 'P') {
            const offCount = secEmps.filter(e => ensureEmployeeRecord(period, e.id).attendance[d] === 'O').length;
            const canOffToday = offCount < maxOff && !forbiddenDays.includes(period.days[d].dow);
            rec.attendance[d] = canOffToday ? 'O' : 'M';
            fixedCount++;
          }
        }
      });

      // 3. Perbaiki > maxOff per section per hari
      const sections = ['BAR', 'Server', 'Kitchen'];
      sections.forEach(sec => {
        const secEmps = emps.filter(e => e.section === sec);
        for (let d = 0; d < numDays; d++) {
          const offEmps = secEmps.filter(e => ensureEmployeeRecord(period, e.id).attendance[d] === 'O');
          if (offEmps.length > maxOff) {
            for (let i = maxOff; i < offEmps.length; i++) {
              const emp = offEmps[i];
              const rec = ensureEmployeeRecord(period, emp.id);
              const prevCode = d > 0 ? rec.attendance[d - 1] : '';
              const pCount = secEmps.filter(e => ensureEmployeeRecord(period, e.id).attendance[d] === 'P').length;
              rec.attendance[d] = (prevCode !== 'M' && pCount === 0) ? 'P' : 'M';
              fixedCount++;
            }
          }
        }
      });

      saveState();
      renderAttendanceTab();
      showToast(`Berhasil memperbaiki ${fixedCount} konflik jadwal!`);
    }

    // Hapus periode yang sedang dipilih (beserta data jadwal, absensi, lembur, telat, & request off-nya)
    function confirmDeletePeriod() {
      if (appState.periods.length <= 1) { showToast('Minimal harus ada satu periode.'); return; }
      const period = getActivePeriod();
      const onlyHere = appState.employees.filter(e => Array.isArray(e.periods) && e.periods.length > 0 && e.periods.every(id => id === period.id));
      let msg = `Periode "${period.shortLabel}" akan dihapus beserta seluruh jadwal, absensi, lembur, keterlambatan, dan Request Off di dalamnya.`;
      if (onlyHere.length > 0) msg += ` Karyawan yang hanya terdaftar di periode ini juga ikut dihapus: ${onlyHere.map(e => e.name).join(', ')}.`;
      msg += ' Tindakan ini tidak bisa dibatalkan, jadi sebaiknya unduh cadangan JSON dulu.';
      openConfirmModal('Hapus Periode?', msg, () => {
        const idx = appState.periods.findIndex(p => p.id === period.id);
        if (idx < 0) return;
        const orphanIds = new Set(onlyHere.map(e => e.id));
        appState.periods.splice(idx, 1);
        appState.employees = appState.employees.filter(e => !orphanIds.has(e.id));
        appState.employees.forEach(e => {
          if (Array.isArray(e.periods)) e.periods = e.periods.filter(id => id !== period.id);
        });
        const next = appState.periods[Math.max(0, idx - 1)];
        appState.activePeriodId = next.id;
        lastClearSnapshot = null;
        const undoBtn = document.getElementById('undo-clear-btn');
        if (undoBtn) undoBtn.classList.add('hidden');
        saveState();
        renderAll();
        showToast(`Periode "${period.shortLabel}" dihapus. Sekarang di ${next.shortLabel}.`);
      });
    }

    function applyCellStatus(empId, dayIndex, isDrag = false) {
      const period = getActivePeriod();
      const rec = ensureEmployeeRecord(period, empId);
      const dayObj = period.days[dayIndex];
      const isSaturday = dayObj && dayObj.dow === 6;
      const current = rec.attendance[dayIndex] || '';

      if (activeBrush === 'CYCLE') {
        if (isDrag) return;
        // Hari Sabtu tidak boleh Off (O), jadi lewati 'O' saat klik beruntun!
        const order = isSaturday ? CYCLE_ORDER_NO_OFF : CYCLE_ORDER;
        const curPos = order.indexOf(current);
        const nextIdx = curPos === -1 ? 1 : (curPos + 1) % order.length;
        rec.attendance[dayIndex] = order[nextIdx];
      } else {
        if (activeBrush === 'O' && isSaturday) {
          if (!isDrag) {
            showToast('Hari Sabtu tidak diperbolehkan Off (O) untuk semua section!');
          }
          return;
        }
        if (isDrag) {
          rec.attendance[dayIndex] = activeBrush;
        } else {
          rec.attendance[dayIndex] = (current === activeBrush && activeBrush !== '') ? '' : activeBrush;
        }
      }

      saveState();
      updateGlobalBadges();
      renderAttendanceTab();
    }

    function handleCellPointerEnter(empId, dayIndex) {

    function renderCoverageMonitor(period, periodEmps) {
      const sections = ['BAR', 'Server', 'Kitchen'];
      const maxOff = Number(appState.settings.maxDailyOffPerSection) || 1;
      const kitchenPTarget = Number(appState.settings.kitchenPagiTarget) || 1;

      const isDayScheduled = period.days.map((_, dayIdx) =>
        periodEmps.some(emp => {
          const rec = ensureEmployeeRecord(period, emp.id);
          return rec.attendance[dayIdx] && rec.attendance[dayIdx] !== '';
        })
      );

      const dateHeaders = period.days.map(d => {
        const numCls = d.isWeekend ? 'text-orange-400 font-bold' : 'text-slate-300 font-bold';
        const dayCls = d.isWeekend ? 'text-orange-400 font-bold' : 'text-slate-500';
        const bgCls = d.isWeekend ? 'bg-orange-500/10' : 'bg-slate-900';
        return `
          <th class="px-1 py-2 text-center min-w-[40px] border-b border-slate-800 ${bgCls} tabular-nums">
            <div class="text-[9px] leading-none ${dayCls}">${d.weekday}</div>
            <div class="text-xs mt-0.5 ${numCls}">${d.dayNum}</div>
          </th>
        `;
      }).join('');

      let bodyHtml = '';
      sections.forEach(sec => {
        const secEmps = periodEmps.filter(e => e.section === sec);
        const secCls = SECTION_BADGE_CLS[sec] || 'bg-slate-800 text-slate-300';

        const shiftRows = [
          { code: 'P', label: 'Pagi (P)' },
          { code: 'MD', label: 'Middle (MD)' },
          { code: 'M', label: 'Malam (M)' },
          { code: 'O', label: 'Off (O)' }
        ];

        shiftRows.forEach((sr, shiftIdx) => {
          const cells = period.days.map((d, dayIdx) => {
            if (!isDayScheduled[dayIdx]) {
              return `<td class="p-1 text-center border-b border-slate-800/60 text-[11px] text-slate-600">-</td>`;
            }
            const count = secEmps.filter(emp => {
              const rec = ensureEmployeeRecord(period, emp.id);
              return rec.attendance[dayIdx] === sr.code;
            }).length;

            let cellCls = 'bg-emerald-500/15 text-emerald-300 font-semibold border-emerald-500/30';
            if (sr.code === 'MD') {
              cellCls = count > 0
                ? 'bg-teal-500/20 text-teal-300 font-semibold border-teal-500/30'
                : 'bg-slate-950/60 text-slate-600 border-slate-800/80';
            } else if (sr.code === 'O') {
              if ((d.dow === 6 && count > 0) || count > maxOff) {
                cellCls = 'bg-rose-500/25 text-rose-300 font-extrabold border-rose-500/50';
              } else if (count === 1) {
                cellCls = 'bg-slate-800 text-slate-300 font-semibold border-slate-700';
              } else {
                cellCls = 'bg-slate-950/60 text-slate-600 border-slate-800/80';
              }
            } else {
              if (count === 0) cellCls = 'bg-rose-500/20 text-rose-300 font-extrabold border-rose-500/40';
              else if (sec === 'Kitchen' && secEmps.length === 3 && sr.code === 'P' && count !== kitchenPTarget) {
                cellCls = 'bg-amber-500/25 text-amber-300 font-extrabold border-amber-500/50';
              }
            }

            return `
              <td class="p-1 text-center border-b border-slate-800/60 ${d.isWeekend ? 'bg-orange-500/[0.03]' : ''}">
                <span class="inline-flex items-center justify-center w-6 h-6 rounded-md border text-[11px] font-mono tabular-nums ${cellCls}">
                  ${count}
                </span>
              </td>
            `;
          }).join('');

          bodyHtml += `
            <tr class="${shiftIdx === shiftRows.length - 1 ? 'border-b-2 border-slate-800' : ''}">
              ${shiftIdx === 0 ? `
                <td rowspan="${shiftRows.length}" class="sticky left-0 z-10 bg-slate-900 px-3 py-2 border-b-2 border-slate-800 align-middle whitespace-nowrap">
                  <span class="inline-block px-2 py-0.5 rounded-md border text-[10px] font-bold ${secCls}">${sec}</span>
                </td>
              ` : ''}
              <td class="sticky left-[74px] z-10 bg-slate-900 px-3 py-1.5 border-b border-slate-800/60 text-xs font-semibold text-slate-300 whitespace-nowrap sticky-shadow">
                ${sr.label}
              </td>
              ${cells}
            </tr>
          `;
        });
      });

      document.getElementById('coverage-monitor-container').innerHTML = `
        <table class="w-full border-collapse text-left">
          <thead>
            <tr>
              <th class="sticky left-0 z-20 bg-slate-900 px-3 py-2 text-[11px] font-bold uppercase text-slate-400 border-b border-slate-800 min-w-[74px]">Section</th>
              <th class="sticky left-[74px] z-20 bg-slate-900 px-3 py-2 text-[11px] font-bold uppercase text-slate-400 border-b border-slate-800 min-w-[114px] sticky-shadow">Shift</th>
              ${dateHeaders}
            </tr>
          </thead>
          <tbody>
            ${bodyHtml}
          </tbody>
        </table>
      `;
    }

    function setOtLateMode(mode) {
      otLateMode = mode;
      renderOtLateTab();
    }

    function handleOtLateInput(empId, dayIndex, rawValue) {
      const period = getActivePeriod();
      const rec = ensureEmployeeRecord(period, empId);
      const val = Math.max(0, parseFloat(rawValue) || 0);

      if (otLateMode === 'overtime') {
        rec.overtime[dayIndex] = val;
      } else {
        rec.lateness[dayIndex] = Math.round(val);
      }

      saveState();
      updateGlobalBadges();
      renderOtLateTab();
    }

    function renderOtLateTab() {
      const btnOt = document.getElementById('mode-btn-overtime');
      const btnLate = document.getElementById('mode-btn-lateness');
      const banner = document.getElementById('ot-late-info-banner');
      const limit = Number(appState.settings.monthlyLateLimit) || 150;

      if (otLateMode === 'overtime') {
        btnOt.className = 'px-4 py-2 rounded-lg text-xs sm:text-sm font-bold transition cursor-pointer bg-amber-500 text-slate-950 shadow-sm';

      const name = document.getElementById('emp-name').value.trim();

    // Kelola Request Off Karyawan (Non-Sabtu, mengizinkan Jumping M -> P)
    function addRequestOff(fromEditPanel = false) {
      const period = getActivePeriod();
      const empSel = document.getElementById(fromEditPanel ? 'edit-req-off-emp' : 'req-off-emp');
      const daySel = document.getElementById(fromEditPanel ? 'edit-req-off-day' : 'req-off-day');
      const noteEl = document.getElementById(fromEditPanel ? 'edit-req-off-note' : 'req-off-note');
      if (!empSel || !daySel) return;

      const empId = empSel.value;
      const dayIdx = Number(daySel.value);
      const note = (noteEl?.value || '').trim();
      const dayObj = period.days[dayIdx];

      if (!dayObj || dayObj.dow === 6) {
        showToast('Hari Sabtu dilarang untuk Request Off semua section!');
        return;
      }

      if (!Array.isArray(period.requestOffs)) period.requestOffs = [];
      const exists = period.requestOffs.find(r => r.empId === empId && r.dayIdx === dayIdx);
      if (exists) {
        showToast('Karyawan ini sudah terdaftar Request Off pada tanggal tersebut.');
        return;
      }

      period.requestOffs.push({
        id: 'req-' + Date.now(),
        empId,
        dayIdx,
        note
      });

      // Langsung tandai 'O' pada tanggal request tersebut agar terlihat jelas
      const rec = ensureEmployeeRecord(period, empId);
      rec.attendance[dayIdx] = 'O';

      if (noteEl) noteEl.value = '';
      saveState();
      if (activeTab === 'karyawan-jadwal') renderEmployeesAndSchedulerTab();
      if (activeTab === 'edit-panel') renderEditPanelTab();
      const emp = appState.employees.find(e => e.id === empId);
      showToast(`Request Off ${emp?.name || ''} (${dayObj.fullLabel}) ditambahkan (Jumping M→P diizinkan).`);
    }

    function removeRequestOff(reqId) {

      if (activeTab === 'karyawan-jadwal') renderEmployeesAndSchedulerTab();
      if (activeTab === 'edit-panel') renderEditPanelTab();
      showToast('Request Off dihapus.');
    }

    // Analisis Carryover dari Periode Sebelumnya
    function computeCarryoverFromPreviousPeriod(targetPeriod) {
      const idx = appState.periods.findIndex(p => p.id === targetPeriod.id);
      if (idx <= 0) return { hasPrev: false, priorityCarryoverEmps: [], lastShiftMap: {}, prevLabel: '' };

      const prevPeriod = appState.periods[idx - 1];
      const prevLen = prevPeriod.days.length;
      const lastWeekStart = Math.max(0, prevLen - 7);
      const targetEmps = getPeriodEmployees(targetPeriod);

      const priorityCarryoverEmps = [];
      const lastShiftMap = {};

      targetEmps.forEach(emp => {
        const prevRec = prevPeriod.records[emp.id];
        if (!prevRec) {
          priorityCarryoverEmps.push({ id: emp.id, name: emp.name, section: emp.section, reason: 'Anggota baru tim periode ini' });
          lastShiftMap[emp.id] = '';
          return;
        }
        const lastSlice = prevRec.attendance.slice(lastWeekStart, prevLen);
        const hadOffInLastWeek = lastSlice.includes('O');
        const hasEmptyInLastWeek = lastSlice.some(c => !c || c === '');
        lastShiftMap[emp.id] = prevRec.attendance[prevLen - 1] || '';

        if (!hadOffInLastWeek || hasEmptyInLastWeek) {
          priorityCarryoverEmps.push({
            id: emp.id,
            name: emp.name,
            section: emp.section,
            reason: hasEmptyInLastWeek ? 'Jadwal minggu terakhir belum terisi' : 'Belum mendapat Off di minggu terakhir'
          });
        }
      });

      return {
        hasPrev: true,
        prevLabel: prevPeriod.shortLabel,
        priorityCarryoverEmps,
        lastShiftMap
      };
    }

    /**
     * ALGORITMA SMART AUTO-SCHEDULER RENTANG TANGGAL [startDayIdx..endDayIdx]:
     * 1. Hanya mengubah hari d di dalam rentang [startDayIdx..endDayIdx].
     * 2. TIDAK PERNAH mengisi 'MD' (Middle hanya dapat diatur manual).
     * 3. TIDAK PERNAH memberi Off ('O') pada hari Sabtu (dow === 6) untuk SEMUA section.
     * 4. Untuk Server & Kitchen, TIDAK PERNAH memberi Off ('O') pada hari Rabu (dow === 3) & Sabtu (dow === 6).
     * 5. Request Off karyawan di tanggal non-Sabtu otomatis di-set 'O' dan karyawan tersebut DIIZINKAN jumping M -> P jika dibutuhkan untuk coverage.
     * 6. Karyawan tanpa Request Off dilarang keras jumping M -> P (P -> M diperbolehkan).
     * 7. Jatah Off 1x/minggu per karyawan, maks 1 orang Off/section/hari, dan Kitchen (3 staf) Pagi = 1 orang/hari.
     */
    function generateStrictSmartSchedule({ period, targetEmpIds, startDayIdx = 0, endDayIdx, overwriteExisting = false }) {
      const numDays = period.days.length;
      const safeStart = Math.max(0, Math.min(numDays - 1, startDayIdx));
      const safeEnd = Math.max(safeStart, Math.min(numDays - 1, endDayIdx !== undefined ? endDayIdx : numDays - 1));

      const periodEmps = getPeriodEmployees(period);
      const targetSet = new Set(targetEmpIds);
      const carryover = computeCarryoverFromPreviousPeriod(period);
      const carryoverPrioritySet = new Set(carryover.priorityCarryoverEmps.map(x => x.id));

      const maxDailyOff = Number(appState.settings.maxDailyOffPerSection) || 1;
      const kitchenPTarget = Number(appState.settings.kitchenPagiTarget) || 1;
      const requestOffs = period.requestOffs || [];
      const requestOffEmpSet = new Set(requestOffs.map(r => r.empId));

      // Jika overwriteExisting aktif, kosongkan sel target HANYA di dalam rentang [safeStart..safeEnd]
      if (overwriteExisting) {
        targetEmpIds.forEach(empId => {
          const rec = ensureEmployeeRecord(period, empId);
          for (let d = safeStart; d <= safeEnd; d++) {
            rec.attendance[d] = '';
          }
        });
      }

      // Terapkan Request Off terlebih dahulu di dalam rentang [safeStart..safeEnd]
      const requestOffDayMap = {}; // key: `${empId}:${dayIdx}` -> true
      requestOffs.forEach(req => {
        const d = Number(req.dayIdx);
        if (d >= safeStart && d <= safeEnd && period.days[d] && period.days[d].dow !== 6) {
          requestOffDayMap[`${req.empId}:${d}`] = true;
          if (targetSet.has(req.empId)) {
            const rec = ensureEmployeeRecord(period, req.empId);
            rec.attendance[d] = 'O';
          }
        }
      });

      const getPrevShift = (empId, d) => {
        if (d > 0) {
          const rec = ensureEmployeeRecord(period, empId);
          return rec.attendance[d - 1] || '';
        }
        return carryover.lastShiftMap[empId] || '';
      };

      let cellsFilled = 0;
      const sections = ['BAR', 'Server', 'Kitchen'];

      sections.forEach(sec => {
        const secEmps = periodEmps.filter(e => e.section === sec);
        if (secEmps.length === 0) return;

        // Hari yang dilarang Off otomatis untuk section ini (selalu termasuk 6 = Sabtu; default Server/Kitchen juga 3 = Rabu)
        const forbiddenDows = new Set(appState.settings.forbiddenOffDays?.[sec] || [6]);
        forbiddenDows.add(6);

        // Proses per blok minggu (7 hari) yang beririsan dengan [safeStart..safeEnd]
        for (let weekStart = 0; weekStart < numDays; weekStart += 7) {
          const weekEnd = Math.min(numDays - 1, weekStart + 6);
          if (weekEnd < safeStart || weekStart > safeEnd) continue;

          const effectiveStart = Math.max(weekStart, safeStart);
          const effectiveEnd = Math.min(weekEnd, safeEnd);
          const isPartialLastWeek = (weekEnd - weekStart + 1) < 7;

          const offDayForEmp = {};
          const dailyOffCounts = {}; // dayIdx -> count of 'O' in this section

          // Hitung 'O' yang sudah ada dalam minggu ini (baik di luar rentang maupun Request Off)
          for (let d = weekStart; d <= weekEnd; d++) {
            let cnt = 0;
            secEmps.forEach(emp => {
              const rec = ensureEmployeeRecord(period, emp.id);
              if (rec.attendance[d] === 'O') {
                offDayForEmp[emp.id] = d;
                cnt++;
              }
            });
            dailyOffCounts[d] = cnt;
          }

          // Urutkan karyawan untuk alokasi jatah Off mingguan (1x per minggu per karyawan)
          const sortedForOff = [...secEmps].sort((a, b) => {
            if (weekStart === 0) {
              const aCarry = carryoverPrioritySet.has(a.id) ? 1 : 0;
              const bCarry = carryoverPrioritySet.has(b.id) ? 1 : 0;
              if (bCarry !== aCarry) return bCarry - aCarry;
            }
            const getStreakBefore = (empId) => {
              const rec = ensureEmployeeRecord(period, empId);
              let st = 0;
              for (let back = effectiveStart - 1; back >= 0; back--) {
                const c = rec.attendance[back];
                if (c === 'P' || c === 'MD' || c === 'M') st++;
                else break;
              }
              return st;
            };
            return getStreakBefore(b.id) - getStreakBefore(a.id);
          });

          sortedForOff.forEach((emp, rankIdx) => {
            if (!targetSet.has(emp.id)) return;
            if (offDayForEmp[emp.id] !== undefined) return; // Sudah dapat Off di minggu ini

            const rec = ensureEmployeeRecord(period, emp.id);
            let lastOffIdx = -99;
            for (let back = effectiveStart - 1; back >= 0; back--) {
              if (rec.attendance[back] === 'O') {
                lastOffIdx = back;
                break;
              }
            }

            // Kandidat hari di dalam [effectiveStart..effectiveEnd] yang:
            // - BUKAN hari terlarang Off (misal Sabtu untuk semua, Rabu & Sabtu untuk Server/Kitchen)
            // - Jumlah Off section pada hari tersebut < maxDailyOff (default 1)
            const candidateDays = [];
            for (let d = effectiveStart; d <= effectiveEnd; d++) {
              const dow = period.days[d].dow;
              if (forbiddenDows.has(dow)) continue;
              if ((dailyOffCounts[d] || 0) >= maxDailyOff) continue;
              const existingCode = rec.attendance[d];
              if (overwriteExisting || !existingCode || existingCode === '') {
                candidateDays.push(d);
              }
            }

            if (candidateDays.length === 0) return;

            if (isPartialLastWeek) {
              const validPartialDays = candidateDays.filter(d => (d - lastOffIdx) >= 5);
              if (validPartialDays.length === 0) return;
              const chosen = validPartialDays[0];
              offDayForEmp[emp.id] = chosen;
              dailyOffCounts[chosen] = (dailyOffCounts[chosen] || 0) + 1;
              return;
            }

            let bestDay = candidateDays[0];
            let bestScore = -999;
            candidateDays.forEach(d => {
              const gap = lastOffIdx === -99 ? (6 + rankIdx) : (d - lastOffIdx);
              let score = -Math.abs(gap - 6.5);
              const prevShift = getPrevShift(emp.id, d);
              if (prevShift === 'M') score += 1.5;
              if (weekStart === 0 && carryoverPrioritySet.has(emp.id)) {
                score += (effectiveEnd - d) * 0.6;
              }
              if (score > bestScore) {
                bestScore = score;
                bestDay = d;
              }
            });

            offDayForEmp[emp.id] = bestDay;
            dailyOffCounts[bestDay] = (dailyOffCounts[bestDay] || 0) + 1;
          });

          // Isi shift hari demi hari pada [effectiveStart..effectiveEnd]
          for (let d = effectiveStart; d <= effectiveEnd; d++) {
            // 1. Terapkan Off ('O')
            secEmps.forEach(emp => {
              if (!targetSet.has(emp.id)) return;
              const rec = ensureEmployeeRecord(period, emp.id);
              if (requestOffDayMap[`${emp.id}:${d}`]) {
                rec.attendance[d] = 'O';
                cellsFilled++;
                return;
              }
              if (!overwriteExisting && rec.attendance[d] && rec.attendance[d] !== '') return;
              if (offDayForEmp[emp.id] === d) {
                rec.attendance[d] = 'O';
                cellsFilled++;
              }
            });

            // 2. Kumpulkan staf yang belum terisi pada hari d
            const workingToAssign = secEmps.filter(emp => {
              if (!targetSet.has(emp.id)) return false;
              const rec = ensureEmployeeRecord(period, emp.id);
              return !rec.attendance[d] || rec.attendance[d] === '';
            });

            if (workingToAssign.length === 0) continue;

            let currentP = secEmps.filter(e => ensureEmployeeRecord(period, e.id).attendance[d] === 'P').length;
            let currentM = secEmps.filter(e => ensureEmployeeRecord(period, e.id).attendance[d] === 'M').length;

            // Kuota Pagi ('P'):
            // - Khusus Kitchen (3 staf): hanya 1 orang Pagi per hari (kitchenPTarget)
            // - Section lain: seimbang setengah tim aktif (minimal 1)
            let desiredP = 1;
            if (sec === 'Kitchen' && secEmps.length === 3) {
              desiredP = kitchenPTarget;
            } else {
              const totalActiveToday = workingToAssign.length + currentP + currentM;
              desiredP = Math.max(1, Math.floor(totalActiveToday / 2));
            }

            // Kandidat Pagi reguler (tidak jumping M -> P)
            const strictEligibleForP = workingToAssign.filter(emp => getPrevShift(emp.id, d) !== 'M');
            // Kandidat tambahan yang punya Request Off di periode ini (diizinkan jumping M -> P jika dibutuhkan coverage)
            const requestOffJumpEligible = workingToAssign.filter(
              emp => getPrevShift(emp.id, d) === 'M' && requestOffEmpSet.has(emp.id)
            );

            const sortCandidatesP = (list) => list.sort((a, b) => {
              const recA = ensureEmployeeRecord(period, a.id);
              const recB = ensureEmployeeRecord(period, b.id);
              const prevA = getPrevShift(a.id, d);
              const prevB = getPrevShift(b.id, d);

              const pStreak = (rec, day) => {
                let s = 0;
                for (let back = day - 1; back >= 0; back--) {
                  if (rec.attendance[back] === 'P') s++;
                  else break;
                }
                return s;
              };

              const stA = pStreak(recA, d);
              const stB = pStreak(recB, d);

              const keepBlockA = (prevA === 'P' && stA < 3) ? 2 : 0;
              const keepBlockB = (prevB === 'P' && stB < 3) ? 2 : 0;
              if (keepBlockB !== keepBlockA) return keepBlockB - keepBlockA;

              const prefA = a.shiftPreference === 'morning' ? 2 : (a.shiftPreference === 'night' ? -2 : 0);
              const prefB = b.shiftPreference === 'morning' ? 2 : (b.shiftPreference === 'night' ? -2 : 0);
              if (prefB !== prefA) return prefB - prefA;

              const totalPA = recA.attendance.filter(c => c === 'P').length;
              const totalPB = recB.attendance.filter(c => c === 'P').length;
              return totalPA - totalPB;
            });

            sortCandidatesP(strictEligibleForP);
            sortCandidatesP(requestOffJumpEligible);

            const combinedEligibleP = [...strictEligibleForP, ...requestOffJumpEligible];
            const neededP = Math.max(0, desiredP - currentP);
            const assignedToP = new Set();

            for (let i = 0; i < Math.min(neededP, combinedEligibleP.length); i++) {
              const emp = combinedEligibleP[i];
              const rec = ensureEmployeeRecord(period, emp.id);
              rec.attendance[d] = 'P';
              assignedToP.add(emp.id);
              currentP++;
              cellsFilled++;
            }

            // Sisanya masuk Malam ('M') — TIDAK PERNAH mengisi 'MD' otomatis!
            workingToAssign.forEach(emp => {
              if (assignedToP.has(emp.id)) return;
              const rec = ensureEmployeeRecord(period, emp.id);
              rec.attendance[d] = 'M';
              currentM++;
              cellsFilled++;
            });
          }
        }
      });

      return cellsFilled;
    }

    function detectFirstEmptyDayIndex(period) {
      const periodEmps = getPeriodEmployees(period);
      for (let d = 0; d < period.days.length; d++) {
        const hasEmpty = periodEmps.some(emp => {
          const rec = ensureEmployeeRecord(period, emp.id);
          return !rec.attendance[d] || rec.attendance[d] === '';
        });
        if (hasEmpty) return d;
      }
      return 0;
    }

    function syncRangeBounds(changedSide) {
      const startSel = document.getElementById('sched-range-start');
      const endSel = document.getElementById('sched-range-end');
      if (!startSel || !endSel) return;
      const s = Number(startSel.value);
      const e = Number(endSel.value);
      if (changedSide === 'start' && s > e) {
        endSel.value = String(s);
      } else if (changedSide === 'end' && e < s) {
        startSel.value = String(e);
      }
    }

    function applyRangePreset(preset) {
      const period = getActivePeriod();
      const maxIdx = period.days.length - 1;
      const startSel = document.getElementById('sched-range-start');
      const endSel = document.getElementById('sched-range-end');
      if (!startSel || !endSel) return;

      let s = 0;
      let e = maxIdx;
      if (preset === 'first_empty') {
        s = detectFirstEmptyDayIndex(period);
        e = maxIdx;
      } else if (preset === 'w1') {
        s = 0; e = Math.min(6, maxIdx);
      } else if (preset === 'w2') {
        s = Math.min(7, maxIdx); e = Math.min(13, maxIdx);
      } else if (preset === 'w3') {
        s = Math.min(14, maxIdx); e = Math.min(20, maxIdx);
      } else if (preset === 'w4') {
        s = Math.min(21, maxIdx); e = maxIdx;
      }

      startSel.value = String(s);
      endSel.value = String(e);
      showToast(`Rentang dipilih: ${period.days[s].fullLabel} – ${period.days[e].fullLabel}`);
    }

    function getSelectedRangeAndEmps() {
      const period = getActivePeriod();
      const startDayIdx = Number(document.getElementById('sched-range-start')?.value || 0);
      const endDayIdx = Number(document.getElementById('sched-range-end')?.value ?? (period.days.length - 1));
      const scopeSec = document.getElementById('sched-scope-section')?.value || 'ALL';
      const periodEmps = getPeriodEmployees(period);
      const targetEmps = scopeSec === 'ALL' ? periodEmps : periodEmps.filter(e => e.section === scopeSec);
      return { period, startDayIdx, endDayIdx, scopeSec, targetEmps };
    }

    function runRangeAutoScheduler() {
      const { period, startDayIdx, endDayIdx, scopeSec, targetEmps } = getSelectedRangeAndEmps();
      const writeMode = document.querySelector('input[name="sched-write-mode"]:checked')?.value || 'empty_only';
      const overwriteExisting = (writeMode === 'overwrite');

      const filledCount = generateStrictSmartSchedule({
        period,
        targetEmpIds: targetEmps.map(e => e.id),
        startDayIdx,
        endDayIdx,
        overwriteExisting
      });

      saveState();
      updateGlobalBadges();
      renderEmployeesAndSchedulerTab();

      const startLbl = period.days[startDayIdx]?.fullLabel || '';
      const endLbl = period.days[endDayIdx]?.fullLabel || '';
      const resultCard = document.getElementById('scheduler-result-card');
      document.getElementById('scheduler-result-title').textContent = `Jadwal Rentang ${startLbl} – ${endLbl} Selesai (${filledCount} Shift)!`;
      document.getElementById('scheduler-result-desc').textContent =
        `Scope: ${scopeSec === 'ALL' ? 'Semua Section' : scopeSec} • Tanpa Middle (MD), Sabtu bebas Off, Server & Kitchen bebas Rabu/Sabtu Off, Kitchen Pagi 1 orang/hari.`;
      resultCard.classList.remove('hidden');

      showToast(`Jadwal otomatis dibuat untuk ${startLbl} – ${endLbl} (${filledCount} sel)!`);
    }

    function clearScheduleInRange() {
      const { period, startDayIdx, endDayIdx, scopeSec, targetEmps } = getSelectedRangeAndEmps();
      const startLbl = period.days[startDayIdx]?.fullLabel || '';
      const endLbl = period.days[endDayIdx]?.fullLabel || '';

      openConfirmModal(
        'Hapus Jadwal pada Rentang Terpilih?',
        `Jadwal untuk ${scopeSec === 'ALL' ? 'Semua Section' : 'Section ' + scopeSec} dari tanggal ${startLbl} sampai ${endLbl} akan dikosongkan. Anda dapat membatalkannya dengan tombol Undo.`,
        () => {
          // Simpan snapshot untuk fitur Undo
          lastClearSnapshot = {
            periodId: period.id,
            records: JSON.parse(JSON.stringify(period.records)),
            label: `${startLbl} – ${endLbl}`
          };

          let cleared = 0;
          targetEmps.forEach(emp => {
            const rec = ensureEmployeeRecord(period, emp.id);
            for (let d = startDayIdx; d <= endDayIdx; d++) {
              if (rec.attendance[d] !== '') {
                rec.attendance[d] = '';
                cleared++;
              }
            }
          });

          saveState();
          updateGlobalBadges();
          renderEmployeesAndSchedulerTab();
          const undoBtn = document.getElementById('undo-clear-btn');
          if (undoBtn) undoBtn.classList.remove('hidden');
          showToast(`${cleared} sel jadwal pada ${startLbl} – ${endLbl} dihapus. Klik Undo jika ingin membatalkan.`);
        }
      );
    }

    function undoLastClear() {
      if (!lastClearSnapshot) return;
      const period = appState.periods.find(p => p.id === lastClearSnapshot.periodId);
      if (!period) return;
      period.records = lastClearSnapshot.records;
      const restoredLabel = lastClearSnapshot.label;
      lastClearSnapshot = null;

      saveState();
      updateGlobalBadges();
      renderEmployeesAndSchedulerTab();
      const undoBtn = document.getElementById('undo-clear-btn');
      if (undoBtn) undoBtn.classList.add('hidden');
      showToast(`Jadwal ${restoredLabel} berhasil dipulihkan (Undo)!`);
    }

    function runRolloverToNextPeriod() {
      const curPeriod = getActivePeriod();
      const curIdx = appState.periods.findIndex(p => p.id === curPeriod.id);
      const nextPeriod = appState.periods[curIdx + 1];
      if (!nextPeriod) {
        showToast('Belum ada periode berikutnya. Buat periode baru terlebih dahulu.');
        return;
      }

      // Lengkapi tanggal kosong di periode berjalan terlebih dahulu jika masih ada
      const curEmps = getPeriodEmployees(curPeriod);
      if (curEmps.some(e => ensureEmployeeRecord(curPeriod, e.id).attendance.includes(''))) {
        generateStrictSmartSchedule({
          period: curPeriod,
          targetEmpIds: curEmps.map(e => e.id),
          startDayIdx: 0,
          endDayIdx: curPeriod.days.length - 1,
          overwriteExisting: false
        });
      }

      appState.activePeriodId = nextPeriod.id;
      const nextEmps = getPeriodEmployees(nextPeriod);
      const filled = generateStrictSmartSchedule({
        period: nextPeriod,
        targetEmpIds: nextEmps.map(e => e.id),
        startDayIdx: 0,
        endDayIdx: nextPeriod.days.length - 1,
        overwriteExisting: true
      });

      saveState();
      renderPeriodSelect();
      updateGlobalBadges();
      renderEmployeesAndSchedulerTab();
      showToast(`Jadwal ${nextPeriod.shortLabel} dibuat (${filled} shift) dengan carryover minggu terakhir!`);
    }

    function renderEmployeesAndSchedulerTab() {
      const period = getActivePeriod();
      const periodEmps = getPeriodEmployees(period);
      document.getElementById('roster-badge-count').textContent = `${periodEmps.length} Staf`;

      // Populate dropdown Rentang Tanggal (Dari & Sampai)
      const startSel = document.getElementById('sched-range-start');
      const endSel = document.getElementById('sched-range-end');
      const firstEmptyIdx = detectFirstEmptyDayIndex(period);

      const dayOptionsHtml = (selectedIdx) => period.days.map((d, idx) => `
        <option value="${idx}" ${idx === selectedIdx ? 'selected' : ''}>
          ${d.fullLabel} (${d.weekday}${d.isWeekend ? ' • Weekend' : ''})
        </option>
      `).join('');

      startSel.innerHTML = dayOptionsHtml(firstEmptyIdx);
      endSel.innerHTML = dayOptionsHtml(period.days.length - 1);

      // Populate Request Off dropdowns (hanya tanggal non-Sabtu!)
      const reqEmpSel = document.getElementById('req-off-emp');
      const reqDaySel = document.getElementById('req-off-day');
      if (reqEmpSel) {
        reqEmpSel.innerHTML = periodEmps.map(e => `<option value="${e.id}">${e.name} (${e.section})</option>`).join('');
      }
      if (reqDaySel) {
        reqDaySel.innerHTML = period.days
          .map((d, idx) => ({ d, idx }))
          .filter(item => item.d.dow !== 6) // Sabtu dilarang Request Off
          .map(item => `<option value="${item.idx}">${item.d.fullLabel} (${item.d.weekday})</option>`)
          .join('');
      }

      // Render Daftar Request Off
      const reqList = period.requestOffs || [];
      const reqBadge = document.getElementById('req-off-count-badge');
      const reqContainer = document.getElementById('req-off-list');
      if (reqBadge) reqBadge.textContent = `${reqList.length} Request`;
      if (reqContainer) {
        if (reqList.length === 0) {
          reqContainer.innerHTML = `<span class="text-[11px] text-slate-500">Belum ada Request Off pada periode ini.</span>`;
        } else {
          reqContainer.innerHTML = reqList.map(r => {
            const emp = appState.employees.find(e => e.id === r.empId);
            const dayObj = period.days[r.dayIdx];
            if (!emp || !dayObj) return '';
            return `
              <span class="inline-flex items-center gap-1.5 px-2.5 py-1 rounded-lg bg-amber-500/15 border border-amber-500/35 text-xs text-amber-200">
                <strong>${emp.name}</strong> • ${dayObj.fullLabel} (${dayObj.weekday})
                ${r.note ? `<span class="text-amber-300/75 italic">(${r.note})</span>` : ''}
                <button type="button" onclick="removeRequestOff('${r.id}')" class="ml-1 text-rose-400 hover:text-rose-200 font-bold cursor-pointer" title="Hapus Request Off">✕</button>
              </span>
            `;
          }).join('');
        }
      }

      // Render Preview Carryover & Tombol Rollover ke Bulan Berikutnya
      const curIdx = appState.periods.findIndex(p => p.id === period.id);
      const nextPeriod = appState.periods[curIdx + 1];
      const carryoverBox = document.getElementById('carryover-preview-box');
      if (nextPeriod) {
        const co = computeCarryoverFromPreviousPeriod(nextPeriod);
        const listText = co.priorityCarryoverEmps.length > 0
          ? co.priorityCarryoverEmps.map(x => `<span class="inline-block px-1.5 py-0.5 rounded bg-slate-900 text-amber-200 border border-amber-500/30 font-semibold">${x.name} (${x.section})</span>`).join(' ')
          : '<span class="text-slate-300">Seluruh staf akan disinkronkan dari shift terakhir akhir bulan.</span>';
        carryoverBox.innerHTML = `
          <div class="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-2">
            <div>
              <div class="font-bold text-amber-300">Rollover Minggu Terakhir → ${nextPeriod.shortLabel}:</div>
              <div class="text-amber-100/90 mt-0.5">Karyawan yang belum dapat jadwal/Off di minggu terakhir otomatis diprioritaskan di Minggu 1 bulan berikutnya:</div>
            </div>
            <button type="button" onclick="runRolloverToNextPeriod()" class="shrink-0 px-3 py-2 rounded-xl bg-amber-500 hover:bg-amber-400 text-slate-950 text-xs font-extrabold transition cursor-pointer">
              Generate ${nextPeriod.shortLabel} →
            </button>
          </div>
          <div class="flex flex-wrap gap-1 pt-0.5">${listText}</div>
        `;

    function toggleForbiddenOffDay(sec, dow) {
      if (dow === 6) {
        showToast('Hari Sabtu terkunci sebagai hari terlarang Off untuk semua section.');
        return;
      }
      const arr = appState.settings.forbiddenOffDays[sec] || [6];
      if (arr.includes(dow)) {
        appState.settings.forbiddenOffDays[sec] = arr.filter(d => d !== dow);
      } else {
        appState.settings.forbiddenOffDays[sec] = [...arr, dow];
      }
      saveState();
      renderEditPanelTab();
      showToast(`Hari terlarang Off otomatis untuk ${sec} diperbarui.`);
    }

    function saveJadwalPanelSettings() {
      appState.settings.shiftHours.P = document.getElementById('cfg-hour-P').value.trim() || '08.30–17.30';
      appState.settings.shiftHours.MD = document.getElementById('cfg-hour-MD').value.trim() || '13.00–22.00';
      appState.settings.shiftHours.M = document.getElementById('cfg-hour-M').value.trim() || '17.00–02.00';
      appState.settings.maxDailyOffPerSection = Math.max(1, Number(document.getElementById('cfg-max-daily-off').value) || 1);
      appState.settings.kitchenPagiTarget = Math.max(1, Number(document.getElementById('cfg-kitchen-pagi').value) || 1);
      saveState();
      showToast('Pengaturan Panel Jadwal & Shift berhasil disimpan!');
    }

    function saveLemburPanelSettings() {
      appState.settings.overtimeRate = Math.max(0, Number(document.getElementById('cfg-ot-rate').value) || 0);
      saveState();
      showToast(`Tarif lembur diperbarui menjadi ${formatRupiah(appState.settings.overtimeRate)} / jam!`);
    }

    function saveTelatPanelSettings() {
      appState.settings.monthlyLateLimit = Math.max(1, Number(document.getElementById('cfg-late-limit').value) || 150);
      const maxInputs = document.querySelectorAll('.tier-max-input');
      const fineInputs = document.querySelectorAll('.tier-fine-input');
      const updatedTiers = [];
      maxInputs.forEach((inp, idx) => {
        const maxM = Math.max(1, Number(inp.value) || 10);
        const fine = Math.max(0, Number(fineInputs[idx]?.value) || 0);
        updatedTiers.push({ maxMinutes: maxM, fine });
      });
      updatedTiers.sort((a, b) => a.maxMinutes - b.maxMinutes);
      appState.settings.lateTiers = updatedTiers;
      saveState();
      updateGlobalBadges();
      renderEditPanelTab();
      showToast('Aturan batas & jenjang denda keterlambatan berhasil disimpan!');
    }

    function addLateTierRow() {
      const tiers = appState.settings.lateTiers || [];
      const lastMax = tiers.length > 0 ? Number(tiers[tiers.length - 1].maxMinutes) : 60;
      const newMax = lastMax >= 999 ? 90 : lastMax + 30;
      tiers.push({ maxMinutes: newMax, fine: 75000 });
      tiers.sort((a, b) => a.maxMinutes - b.maxMinutes);
      appState.settings.lateTiers = tiers;
      saveState();
      renderEditPanelTab();
      showToast('Baris jenjang denda baru ditambahkan.');
    }

    function removeLateTierRow(idx) {
      if ((appState.settings.lateTiers || []).length <= 1) {
        showToast('Minimal harus ada 1 jenjang denda keterlambatan.');
        return;
      }
      appState.settings.lateTiers.splice(idx, 1);
      saveState();
      renderEditPanelTab();
      showToast('Baris jenjang denda dihapus.');
    }

    function openNewPeriodModal() {
      // Isi pilihan periode sumber (default: periode terakhir secara kronologis) & saran tanggal berikutnya
      const sorted = [...appState.periods].sort((a, b) => a.startDate < b.startDate ? -1 : a.startDate > b.startDate ? 1 : 0);
      const last = sorted[sorted.length - 1];
      document.getElementById('new-period-source').innerHTML = sorted.map(p =>
        `<option value="${p.id}" ${p.id === last.id ? 'selected' : ''}>${p.shortLabel} — ${getPeriodEmployees(p).length} karyawan</option>`
      ).join('');
      const [ly, lm, ld] = last.endDate.split('-').map(Number);
      const sugStart = new Date(Date.UTC(ly, lm - 1, ld + 1));
      const sugEnd = new Date(Date.UTC(ly, lm - 1, ld + last.days.length));
      document.getElementById('new-period-start').value = sugStart.toISOString().slice(0, 10);
      document.getElementById('new-period-end').value = sugEnd.toISOString().slice(0, 10);
      const err = document.getElementById('new-period-error');
      if (err) err.classList.add('hidden');
      const modal = document.getElementById('new-period-modal');
      modal.classList.remove('hidden');
      modal.classList.add('flex');
    }

    // Jika tanggal mulai diubah dan tanggal selesai jadi tidak valid, selesai disesuaikan otomatis
    function onNewPeriodStartChange() {
      const s = document.getElementById('new-period-start').value;
      const e = document.getElementById('new-period-end').value;
      if (!s) return;
      if (!e || e < s) {
        const sorted = [...appState.periods].sort((a, b) => a.startDate < b.startDate ? -1 : 1);
        const len = sorted.length ? sorted[sorted.length - 1].days.length : 30;
        const [y, m, d] = s.split('-').map(Number);
        document.getElementById('new-period-end').value = new Date(Date.UTC(y, m - 1, d + len - 1)).toISOString().slice(0, 10);
      }
    }

    function handleCreatePeriod(e) {
      try {
        return handleCreatePeriodInner(e);
      } catch (err) {
        console.error('Gagal membuat periode:', err);
        showNewPeriodError('Gagal membuat periode: ' + (err && err.message ? err.message : err));
      }

    function handleCreatePeriodInner(e) {
      e.preventDefault();
      const name = document.getElementById('new-period-name').value.trim();
      const startDate = document.getElementById('new-period-start').value;
      const endDate = document.getElementById('new-period-end').value;
      const divisor = Number(document.getElementById('new-period-divisor').value) || 30;
      const autoSched = document.getElementById('new-period-autosched').checked;

      const days = buildDaysArray(startDate, endDate);
      if (!startDate || !endDate) { showNewPeriodError('Tanggal mulai dan tanggal selesai harus diisi lengkap (tanggal, bulan, dan tahun).'); return; }
      if (endDate < startDate) { showNewPeriodError('Tanggal selesai harus sama dengan atau setelah tanggal mulai.'); return; }
      if (days.length === 0 || days.length > 62) {
        showNewPeriodError(`Rentang periode maksimal 62 hari (sekarang ${days.length} hari).`);
        return;
      }

      const startLbl = days[0].fullLabel;
      const endLbl = days[days.length - 1].fullLabel;
      const newPeriod = {
        id: 'period-' + Date.now(),
        label: `${name} (${startLbl} – ${endLbl} • ${days.length} Hari)`,
        shortLabel: `${name} (${startLbl} – ${endLbl})`,
        startDate,
        endDate,
        days,
        workdayDivisor: divisor,
        records: {},
        requestOffs: []
      };

      // Karyawan & section disalin dari periode sumber yang dipilih (bukan sekadar periode aktif)
      const srcId = document.getElementById('new-period-source').value;
      const srcPeriod = appState.periods.find(p => p.id === srcId) || getActivePeriod();
      const activeEmps = getPeriodEmployees(srcPeriod);
      activeEmps.forEach(emp => {
        if (emp.periods && !emp.periods.includes(newPeriod.id)) emp.periods.push(newPeriod.id);
        ensureEmployeeRecord(newPeriod, emp.id);
      });

      appState.periods.push(newPeriod);
      // Urutkan kronologis agar carryover & "periode berikutnya" selalu benar
      appState.periods.sort((a, b) => a.startDate < b.startDate ? -1 : a.startDate > b.startDate ? 1 : 0);

      if (autoSched) {
        generateStrictSmartSchedule({
          period: newPeriod,
          targetEmpIds: activeEmps.map(emp => emp.id),
          startDayIdx: 0,
          endDayIdx: newPeriod.days.length - 1,
          overwriteExisting: true
        });
      }

      appState.activePeriodId = newPeriod.id;
      saveState();
      closeNewPeriodModal();
      renderAll();
      showToast(`Periode "${name}" berhasil dibuat!`);
    }

    // Kunci tanggal (YYYY-MM-DD) dihitung dari startDate periode agar tidak terpengaruh zona waktu
    function sxDayKey(period, idx) {
      const [y, m, d] = period.startDate.split('-').map(Number);
      return new Date(Date.UTC(y, m - 1, d + idx)).toISOString().slice(0, 10);
    }

    function sxAllKeys() {

    function setScheduleExportPreset(kind) {
      const keys = sxAllKeys();
      const period = getActivePeriod();
      let from = sxDayKey(period, 0), to = sxDayKey(period, period.days.length - 1);
      if (kind === 'week1') to = sxDayKey(period, Math.min(6, period.days.length - 1));
      if (kind === 'all') { from = keys[0]; to = keys[keys.length - 1]; }
      if (kind === 'next7') {
        const t = new Date();
        const today = new Date(Date.UTC(t.getFullYear(), t.getMonth(), t.getDate()));
        from = today.toISOString().slice(0, 10);
        to = new Date(today.getTime() + 6 * 86400000).toISOString().slice(0, 10);
      }
      document.getElementById('sx-from').value = from;
      document.getElementById('sx-to').value = to;
      updateScheduleExportInfo();
    }

    function buildScheduleExportData() {
      const from = document.getElementById('sx-from').value;
      const to = document.getElementById('sx-to').value;
      const sections = sxSelectedSections();
      const dayMap = new Map();
      const emps = new Map();
      appState.periods.forEach(p => {
        const pEmps = getPeriodEmployees(p);
        p.days.forEach((d, i) => {
          const key = sxDayKey(p, i);
          if (key < from || key > to) return;
          let e = dayMap.get(key);
          if (!e) { e = { day: d, key, codes: {} }; dayMap.set(key, e); }
          pEmps.forEach(emp => {
            const rec = p.records[emp.id];
            e.codes[emp.id] = rec ? (rec.attendance[i] || '') : '';
            emps.set(emp.id, emp);
          });
        });
      });
      const days = [...dayMap.values()].sort((a, b) => a.key < b.key ? -1 : 1);
      const order = ['BAR', 'Server', 'Kitchen'];
      const rows = [...emps.values()]
        .filter(e => sections.includes(e.section))
        .sort((a, b) => order.indexOf(a.section) - order.indexOf(b.section) || appState.employees.indexOf(a) - appState.employees.indexOf(b));
      return { from, to, days, rows };
    }

    function updateScheduleExportInfo() {
      const el = document.getElementById('sx-info');
      const data = buildScheduleExportData();
      if (!data.from || !data.to || data.from > data.to) {
        el.textContent = 'Rentang tanggal belum valid.';
      } else if (data.days.length === 0 || data.rows.length === 0) {
        el.textContent = 'Tidak ada data jadwal pada rentang/section ini.';
      } else {
        let blank = 0; const blankDays = new Set();
        data.days.forEach(e => data.rows.forEach(r => { if (!e.codes[r.id]) { blank++; blankDays.add(e.key); } }));
        const total = data.days.length * data.rows.length;
        el.textContent = `${sxFmtDate(data.days[0].key)} – ${sxFmtDate(data.days[data.days.length - 1].key)} • ${data.days.length} hari • ${data.rows.length} karyawan akan diexport.`;
        if (blank > 0) {
          const warn = document.createElement('div');
          warn.className = 'mt-1.5 text-amber-300';
          warn.textContent = (blank === total ? 'Semua kotak jadwal pada rentang ini masih kosong' : `${blank} dari ${total} kotak jadwal masih kosong (${blankDays.size} tanggal)`) + ', jadi shift P/M/O tidak muncul di bagian itu. Isi dulu jadwalnya lewat Auto-Jadwal. ';
          const btn = document.createElement('button');
          btn.type = 'button';
          btn.className = 'underline font-bold cursor-pointer';
          btn.textContent = 'Buka Auto-Jadwal';
          btn.onclick = () => { closeScheduleExport(); openAutoSchedulerQuick(); };
          warn.appendChild(btn);
          el.appendChild(warn);
        }
      }
    }

    function sxFmtDate(key) {
      const [y, m, d] = key.split('-').map(Number);
      const mon = ['Jan', 'Feb', 'Mar', 'Apr', 'Mei', 'Jun', 'Jul', 'Agu', 'Sep', 'Okt', 'Nov', 'Des'][m - 1];
      return `${d} ${mon} ${y}`;
    }

        g.textAlign = 'left';
        [['P', `Pagi (${hrs.P || ''})`], ['MD', `Middle (${hrs.MD || ''})`], ['M', `Malam (${hrs.M || ''})`], ['O', 'Off'], ['S', 'Sakit'], ['I', 'Izin'], ['A', 'Alpa']].forEach(([k, label]) => {

        });
        return { canvas: c, w: W2, h: H };
      });

    // Simpan file: lewat kapabilitas "downloads" di claude.ai (unduh langsung diblokir di halaman terbit),
    // dengan cadangan unduh biasa jika aplikasi dibuka di luar claude.ai
    async function ztdSave(data, filename) {
      let dl = null;
      try { if (window.claude && typeof window.claude.use === 'function') dl = await window.claude.use('downloads'); } catch (e) { dl = null; }
      if (dl) {
        try { await dl.save({ filename, data }); return true; }
        catch (e) {
          if (!e || e.code !== 'declined') { console.error('Simpan file gagal:', e); showToast('Gagal menyimpan file'); }
          return false;
        }
      }
      const blob = data instanceof Blob ? data : new Blob([data]);
      const a = document.createElement('a');
      a.href = URL.createObjectURL(blob);
      a.download = filename;
      document.body.appendChild(a); a.click(); a.remove();
      setTimeout(() => URL.revokeObjectURL(a.href), 2000);
      return true;
    }

    function runScheduleExport(fmt) {
      const data = buildScheduleExportData();
      if (!data.from || !data.to || data.from > data.to) { showToast('Rentang tanggal belum valid'); return; }
      if (data.days.length === 0 || data.rows.length === 0) { showToast('Tidak ada data jadwal untuk diexport'); return; }
      if (data.days.every(e => data.rows.every(r => !e.codes[r.id]))) { showToast('Jadwal pada rentang ini masih kosong. Isi dulu lewat Auto-Jadwal.'); return; }
      const blocks = drawScheduleBlocks(data);
      const dmy = k => k.split('-').reverse().join('-');
      const base = `Jadwal_Karyawan_${dmy(data.days[0].key)}_sd_${dmy(data.days[data.days.length - 1].key)}`;
      if (fmt === 'png') {
        const totalH = blocks.reduce((a, b) => a + b.canvas.height, 0);
        const totalW = Math.max(...blocks.map(b => b.canvas.width));
        const out = document.createElement('canvas');
        out.width = totalW; out.height = totalH;
        const g = out.getContext('2d');
        g.fillStyle = '#ffffff'; g.fillRect(0, 0, totalW, totalH);
        let y = 0;
        blocks.forEach(b => { g.drawImage(b.canvas, 0, y); y += b.canvas.height; });
        out.toBlob(blob => { sxDownload(blob, base + '.png').then(ok => { if (ok) showToast('Gambar jadwal berhasil diunduh'); }); }, 'image/png');
      } else {
        if (!window.jspdf) { showToast('Library PDF belum termuat, coba lagi'); return; }
        const { jsPDF } = window.jspdf;
        const pdf = new jsPDF({ orientation: 'landscape', unit: 'pt', format: 'a4' });
        blocks.forEach((b, i) => {
          if (i > 0) pdf.addPage();
          const pw = pdf.internal.pageSize.getWidth() - 40, ph = pdf.internal.pageSize.getHeight() - 40;
          const sc = Math.min(pw / b.w, ph / b.h);
          pdf.addImage(b.canvas.toDataURL('image/jpeg', 0.95), 'JPEG', 20, 20, b.w * sc, b.h * sc);
        });
        sxDownload(pdf.output('blob'), base + '.pdf').then(ok => { if (ok) showToast('PDF jadwal berhasil diunduh'); });
      }
    }

    function handleBrandLogoFile(input) {
      const file = input.files && input.files[0];
      if (!file) return;
      if (!file.type.startsWith('image/')) { showToast('File harus berupa gambar.'); return; }
      const reader = new FileReader();
      reader.onload = () => {
        const img = new Image();
        img.onload = () => {
          // Perkecil ke maks 256px agar muat di penyimpanan browser
          const max = 256;
          const scale = Math.min(1, max / Math.max(img.width || max, img.height || max));
          const w = Math.max(1, Math.round((img.width || max) * scale));
          const h = Math.max(1, Math.round((img.height || max) * scale));
          const canvas = document.createElement('canvas');
          canvas.width = w; canvas.height = h;
          canvas.getContext('2d').drawImage(img, 0, 0, w, h);
          try {
            brandDraftLogo = canvas.toDataURL('image/png');
          } catch (e) {
            brandDraftLogo = reader.result;
          }
          updateBrandPreview();
        };
        img.onerror = () => showToast('Gagal membaca gambar logo.');
        img.src = reader.result;
      };
      reader.readAsDataURL(file);
    }

    // Pemilih tanggal berformat Tanggal – Bulan – Tahun (menggantikan input date bawaan browser)
    function initDateSelects() {
      const months = ['Januari', 'Februari', 'Maret', 'April', 'Mei', 'Juni', 'Juli', 'Agustus', 'September', 'Oktober', 'November', 'Desember'];
      const selCls = 'min-w-0 px-2 py-2 rounded-xl bg-slate-950 border border-slate-800 text-xs text-slate-100 focus:outline-none focus:border-amber-500/60 cursor-pointer';
      document.querySelectorAll('input[type="date"]').forEach(inp => {
        let cur = inp.value || '';
        inp.type = 'hidden';
        const wrap = document.createElement('div');
        wrap.className = 'flex gap-1.5' + (inp.classList.contains('mt-1') ? ' mt-1' : '');
        const mk = (label, flex) => {
          const s = document.createElement('select');
          s.className = selCls + ' ' + flex;
          s.setAttribute('aria-label', label);
          s.innerHTML = `<option value="">${label}</option>`;
          return s;
        };
        const sd = mk('Tanggal', 'flex-[0.9]'), sm = mk('Bulan', 'flex-[1.7]'), sy = mk('Tahun', 'flex-[1.2]');
        for (let d = 1; d <= 31; d++) sd.insertAdjacentHTML('beforeend', `<option value="${d}">${d}</option>`);
        months.forEach((n, i) => sm.insertAdjacentHTML('beforeend', `<option value="${i + 1}">${n}</option>`));
        const thisYear = new Date().getFullYear();
        const y0 = Math.min(2024, thisYear - 1), y1 = thisYear + 10;
        for (let y = y0; y <= y1; y++) sy.insertAdjacentHTML('beforeend', `<option value="${y}">${y}</option>`);
        wrap.append(sd, sm, sy);
        inp.insertAdjacentElement('afterend', wrap);
        const render = () => {
          const mm = /^(\d{4})-(\d{2})-(\d{2})$/.exec(cur);
          if (mm && ![...sy.options].some(o => o.value === String(Number(mm[1])))) {
            sy.insertAdjacentHTML('beforeend', `<option value="${Number(mm[1])}">${Number(mm[1])}</option>`);
          }
          sd.value = mm ? String(Number(mm[3])) : '';
          sm.value = mm ? String(Number(mm[2])) : '';
          sy.value = mm ? String(Number(mm[1])) : '';
        };
        const onPick = () => {
          if (sd.value && sm.value && sy.value) {
            const dim = new Date(Date.UTC(Number(sy.value), Number(sm.value), 0)).getUTCDate();
            const day = Math.min(Number(sd.value), dim);
            sd.value = String(day);
            cur = `${sy.value}-${String(sm.value).padStart(2, '0')}-${String(day).padStart(2, '0')}`;
          } else {
            cur = '';
          }
          inp.dispatchEvent(new Event('change', { bubbles: true }));
        };
        [sd, sm, sy].forEach(s => s.addEventListener('change', onPick));
        // Kode lain tetap memakai inp.value (format ISO) — getter/setter menyelaraskan dengan pemilih
        Object.defineProperty(inp, 'value', {
          configurable: true,
          get() { return cur; },
          set(v) { cur = v || ''; render(); }
        });
        render();
      });
    }

    const CLOUD = { db: null, canWrite: false, ready: false, last: {}, timer: null, busy: false };

    function setCloudStatus(text, tone) {
      const t = document.getElementById('cloud-text');

    function cloudWithTimeout(promise, ms) {
      return Promise.race([promise, new Promise((_, rej) => setTimeout(() => rej(new Error('timeout')), ms))]);
    }

    // Pecah state menjadi dokumen kecil: satu dokumen per periode + meta + brand (batas 256 KB per dokumen)
    function cloudBuildDocs() {
      const clean = o => JSON.parse(JSON.stringify(o));
      const docs = {};
      appState.periods.forEach(p => { docs['periods/' + p.id] = clean(p); });
      const brand = getBrand();
      if (JSON.stringify(brand).length < 200000) docs['app/brand'] = brand;
      const { periods, ...rest } = appState;
      docs['app/meta'] = clean({ ...rest, periodIds: periods.map(p => p.id), v: 1 });
      return docs; // meta paling akhir agar tidak menunjuk periode yang belum tertulis
    }

    function scheduleCloudSync() {
      if (!CLOUD.db || !CLOUD.canWrite || !CLOUD.ready) return;
      clearTimeout(CLOUD.timer);
      setCloudStatus('Menyimpan ke database…', 'busy');
      CLOUD.timer = setTimeout(cloudSync, 1200);
    }

    async function cloudSync() {
      if (!CLOUD.db || !CLOUD.canWrite) return;
      if (CLOUD.busy) { CLOUD.timer = setTimeout(cloudSync, 800); return; }
      CLOUD.busy = true;
      try {
        const docs = cloudBuildDocs();
        for (const [path, data] of Object.entries(docs)) {
          const str = JSON.stringify(data);
          if (CLOUD.last[path] === str) continue;
          await CLOUD.db.doc(path).set(path === 'app/meta' ? { ...data, savedAt: Date.now() } : data);
          CLOUD.last[path] = str;
        }
        // Hapus dokumen periode yang sudah tidak ada di aplikasi
        for (const path of Object.keys(CLOUD.last)) {
          if (path.startsWith('periods/') && !(path in docs)) {
            await CLOUD.db.doc(path).delete();
            delete CLOUD.last[path];
          }
        }
        setCloudStatus('Tersimpan di database', 'ok');
      } catch (e) {
        console.error('Sinkron database gagal:', e);
        setCloudStatus('Gagal simpan ke database (tersimpan lokal)', 'err');
      }
      CLOUD.busy = false;
    }

    async function cloudInit() {
      try {
        if (!window.claude || typeof window.claude.use !== 'function') { setCloudStatus('Mode lokal (tanpa database)', 'off'); return; }
        const user = await window.claude.use('user');
        const db = await window.claude.use('db');
        if (!db) { setCloudStatus('Mode lokal (database tidak tersedia)', 'off'); return; }
        CLOUD.db = db;
        let can = true;
        if (user && typeof user.can === 'function') { const r = await user.can('data.write'); can = r !== false; }
        CLOUD.canWrite = can;

        const metaSnap = await cloudWithTimeout(db.doc('app/meta').get(), 10000);
        if (metaSnap.exists) {
          const meta = metaSnap.data();
          const ids = Array.isArray(meta.periodIds) ? meta.periodIds : [];
          const snaps = await cloudWithTimeout(Promise.all(ids.map(id => db.doc('periods/' + id).get())), 15000);
          const periods = snaps.filter(sn => sn.exists).map(sn => sn.data());
          if (periods.length !== ids.length) {
            CLOUD.canWrite = false; // jangan menimpa database jika data tidak lengkap
            setCloudStatus('Data database tidak lengkap (mode baca saja)', 'err');
            return;
          }
          const { periodIds, savedAt, v, ...rest } = meta;
          appState = normalizeState({ ...rest, periods });
          const brandSnap = await db.doc('app/brand').get();
          if (brandSnap.exists) { try { localStorage.setItem(BRAND_KEY, JSON.stringify(brandSnap.data())); } catch (e) { /* abaikan */ } }
          try { localStorage.setItem(STORAGE_KEY, JSON.stringify(appState)); } catch (e) { /* abaikan */ }
          const docs = cloudBuildDocs();
          Object.entries(docs).forEach(([k, d]) => { CLOUD.last[k] = JSON.stringify(d); });
          CLOUD.ready = true;
          applyBrand();
          renderAll();
          setCloudStatus(CLOUD.canWrite ? 'Tersinkron dengan database' : 'Database (mode baca saja)', CLOUD.canWrite ? 'ok' : 'off');
        } else if (CLOUD.canWrite) {
          // Database masih kosong: unggah data yang ada di browser ini sebagai data awal
          CLOUD.ready = true;
          setCloudStatus('Mengunggah data awal…', 'busy');
          await cloudSync();
        } else {
          CLOUD.ready = true;
          setCloudStatus('Database kosong (mode baca saja)', 'off');
        }
        CLOUD.ready = true;
      } catch (e) {
        console.error('Database tidak bisa dihubungi:', e);
        setCloudStatus('Mode lokal (database gagal dihubungi)', 'err');
      }
    }
